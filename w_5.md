// Example exec function in OpenClaw environment
const exec = (command: string, workdir?: string): void => {
  if (!workdir) workdir = process.cwd();

  // Execute the command
  execCommand(command, workdir, {
    stdout: process.stdout,
    stderr: process.stderr,
    return: process.exitCode,
  });
};

// Helper function to execute with background continuation
function execCommand(
  command: string,
  workdir?: string,
  {
    stdout: stdoutOptions = { stream: true },
    stderr: stderrOptions = { stream: true },
    return: returnOptions = { exitCode: 0 },
  }: {
    command: string;
    workdir?: string;
    stdout?: boolean;
    stderr?: boolean;
    return?: {
      stdout?: string;
      stderr?: string;
      exitCode?: number;
    };
  } {
    const pty = process.pty || process.stdin;

    // Execute the command in the foreground
    process.exec(command, {
      // Set default options
      stdout: stdoutOptions,
      stderr: stderrOptions,
      return: returnOptions,
      // Use process.stdin for background continuation
      stdin: pty.stdin,
      // Set process environment
      env: {
        ...global.env,
        OPENCLAW_EXEC_PATH: process.execPath,
      },
    });
  }
};
